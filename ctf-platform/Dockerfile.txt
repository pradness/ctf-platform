
FROM ubuntu:22.04

ENV DEBIAN_FRONTEND=noninteractive

COPY setup.sh /setup.sh
RUN sed -i 's/\r$//' /setup.sh && chmod +x /setup.sh && bash /setup.sh

RUN apt-get install -y gosu

RUN useradd -m -s /bin/bash player && \
    echo "player:ctf" | chpasswd

RUN mkdir -p /var/run/sshd

EXPOSE 10000

CMD ["bash", "-c", "PORT=${PORT:-10000}; sed -i \"s/^Listen 80/Listen ${PORT}/\" /etc/apache2/ports.conf; sed -i \"s/<VirtualHost \\*:80>/<VirtualHost *:${PORT}>/\" /etc/apache2/sites-available/000-default.conf; service ssh start; apache2ctl -D FOREGROUND"]
